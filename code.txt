import os, psycopg
from contextlib import contextmanager
from psycopg.rows import dict_row
from dotenv import load_dotenv

load_dotenv()

DB_CONFIG = {
    "host": os.getenv("POSTGRES_HOST", "localhost"),
    "port": int(os.getenv("POSTGRES_PORT", 6543)),
    "dbname": os.getenv("POSTGRES_DB", "museum_db"),
    "user": os.getenv("POSTGRES_USER", "museum"),
    "password": os.getenv("POSTGRES_PASSWORD", "123456789")
}

@contextmanager
def get_connection():
    conn = psycopg.connect(**DB_CONFIG, row_factory=dict_row)
    try:
        yield conn
        conn.commit()
    except Exception:
        conn.rollback()
        raise
    finally:
        conn.close()

def execute(query: str, params: tuple = (), fetch: str | None = None):
    with get_connection() as conn:
        with conn.cursor() as cur:
            cur.execute(query, params)
            if fetch == "one":
                return cur.fetchone()
            if fetch == "all":
                return cur.fetchall()
            return None

def pick_author(authors: Dict[int, Author]) -> Author:
    query = input_str("Введите имя/фамилию автора: ")
    found = find_authors_by_query(authors, query)

    if not found:
        print("Автор не найден. Создаём нового.")
        name = input_str("Имя: ")
        lastname = input_str("Фамилия: ")
        author = create_author(name, lastname)
        author.id = insert_author(author)
        authors[author.id] = author
        return author

    if len(found) == 1:
        return found[0]

    for a in found:
        print_author_info(a)
    author_id = input_int("Введите ID автора: ")
    return authors[author_id]

def pick_collection(collections: Dict[int, Collection]) -> Collection:
    name = input_str("Название коллекции: ")
    coll = find_collection_by_name(collections, name)

def load_authors() -> Dict[int, Author]:
    rows = execute("SELECT id, name, lastname FROM authors ORDER BY id", fetch="all")
    return {r["id"]: Author.from_data(r) for r in rows}

def save_authors(authors: Dict[int, Author]) -> None:
    for a in authors.values():
        execute("""
            INSERT INTO authors (id, name, lastname)
            VALUES (%s, %s, %s)
            ON CONFLICT (id) DO UPDATE SET name = EXCLUDED.name, lastname = EXCLUDED.lastname
            """,
            (a.id, a.name, a.lastname),
        )

def insert_author(a: Author) -> int:
    row = execute(
        "INSERT INTO authors (name, lastname) VALUES (%s, %s) RETURNING id",
        (a.name, a.lastname),
        fetch="one",
    )
    return row["id"]

def load_collections() -> Dict[int, Collection]:
    rows = execute(
        "SELECT id, name, description FROM collections ORDER BY id",
        fetch="all",
    )
    return {r["id"]: Collection.from_data(r) for r in rows}

def insert_collection(c: Collection) -> int:
    row = execute(
        "INSERT INTO collections (name, description) VALUES (%s, %s) RETURNING id",
        (c.name, c.description),
        fetch="one",
    )
    return row["id"]

def load_conditions() -> Dict[int, Condition]:
    rows = execute("""
        SELECT id, code, label, needs_restoration, needs_check
        FROM conditions ORDER BY id
        """,
        fetch="all",
    )
    if not rows:
        return dict(DEFAULT_CONDITIONS)
    return {r["id"]: Condition.from_data(r) for r in rows}

def load_exhibits(authors: Dict[int, Author], collections: Dict[int, Collection], conditions: Dict[int, Condition]) -> Dict[int, Exhibit]:
    rows = execute("""
        SELECT id, name, author_id, collection_id, condition_id, year, price, is_available
        FROM exhibits ORDER BY id
        """,
        fetch="all",
    )
    result: Dict[int, Exhibit] = {}
    for r in rows:
        ex = Exhibit.from_data(r, authors, collections, conditions)
        if ex is not None:
            result[ex.id] = ex
    return result


def insert_exhibit(ex: Exhibit) -> int:
    row = execute("""
        INSERT INTO exhibits(name, author_id, collection_id, condition_id, year, price, is_available)
        VALUES (%s, %s, %s, %s, %s, %s, %s)
        RETURNING id
        """,
        (
            ex.name,
            ex.author.id,
            ex.collection.id,
            ex.condition.id,
            ex.year,
            ex.price,
            ex.is_available,
        ),
        fetch="one",
    )
    return row["id"]

from typing import Any, Dict

class Author:

        self.id = author_id
        self.name = name
        self.lastname = lastname

    def get_full_name(self) -> str:
        return f"{self.name} {self.lastname}"

    def __str__(self) -> str:
        return self.get_full_name()

    @classmethod
    def from_data(cls, data: Dict[str, Any]):
        return cls(
            author_id=data["id"],
            name=data["name"],
            lastname=data["lastname"],
        )

    def to_data(self) -> Dict[str, Any]:
        return {
            "id": self.id,
            "name": self.name,
            "lastname": self.lastname,
        }

def add_author(authors: Dict[int, Author], author_data: Author):
    new_id = max(authors.keys(), default=0) + 1
    author_data.id = new_id
    authors[new_id] = author_data
    return new_id

        self.id = condition_id
        self.code = code
        self.label = label

from datetime import date
from typing import Any, Dict, List
from .authors import Author
from .collections import Collection
from .conditions import Condition

class Exhibit:
    def __init__(self, exhibit_id: int, name: str, author: Author, collection: Collection, condition: Condition, year: int, price: float, is_available: bool):
        self.id = exhibit_id
        self.name = name
        self.author = author
        self.collection = collection
        self.condition = condition
        self.year = year
        self.price = price
        self.is_available = is_available

    def calculate_age(self):
        return date.today().year - self.year

    def get_status(self):
        return self.condition.get_status(self.is_available)

    def __str__(self):
        author_name = (self.author.get_full_name() if self.author else "Неизвестен")
        return (
            f"{self.name} — {author_name}, "
            f"{self.year} г., {self.price} руб."
        )

    @classmethod
    def from_data(cls, data: Dict[str, Any], authors: Dict[int, Author], collections: Dict[int, Collection], conditions: Dict[int, Condition]):
        author = authors.get(data["author_id"])
        collection = collections.get(data["collection_id"])
        condition = conditions.get(data["condition_id"])

        if author is None or collection is None or condition is None:
            return None

        return cls(
            exhibit_id=data["id"],
            name=data["name"],
            author=author,
            collection=collection,
            condition=condition,
            year=data["year"],
            price=data["price"],
            is_available=data["is_available"],
        )

    def to_data(self) -> Dict[str, Any]:
        return {
            "id": self.id,
            "name": self.name,
            "author_id": self.author.id if self.author else None,
            "collection_id": (self.collection.id if self.collection else None),
            "condition_id": (self.condition.id if self.condition else None),
            "year": self.year,
            "price": self.price,
            "is_available": self.is_available,
        }

def create_exhibit(name: str, author: Author, collection: Collection, condition: Condition, year: int, price: float, is_available: bool):
    return Exhibit(
        exhibit_id=0,
        name=name,
        author=author,
        collection=collection,
        condition=condition,
        year=year,
        price=price,
        is_available=is_available,
    )

def add_exhibit(exhibits: Dict[int, Exhibit], exhibit_data: Exhibit):
    new_id = max(exhibits.keys(), default=0) + 1
    exhibit_data.id = new_id
    exhibits[new_id] = exhibit_data
    return new_id


def find_exhibits_by_author(exhibits: Dict[int, Exhibit], authors: Dict[int, Author], query: str):
    q = query.lower()
    matched_ids = {
        aid for aid, a in authors.items()
        if q in a.name.lower() or q in a.lastname.lower()
    }
    return [
        ex for ex in exhibits.values()
        if ex.author and ex.author.id in matched_ids
    ]

def filter_exhibits_by_condition(exhibits: Dict[int, Exhibit], condition_id: int):
    return [ex for ex in exhibits.values() if ex.condition and ex.condition.id == condition_id]

def filter_exhibits_by_collection(exhibits: Dict[int, Exhibit], collection_id: int) -> List[Exhibit]:
    return [ex for ex in exhibits.values() if ex.collection and ex.collection.id == collection_id]

def sort_exhibits_by_price(exhibits: Dict[int, Exhibit]):
    return sorted(exhibits.values(), key=lambda ex: ex.price)

def get_total_price(exhibits: Dict[int, Exhibit]):
    return sum(ex.price for ex in exhibits.values())

def print_exhibit_info(exhibit: Exhibit):
    author_name = (exhibit.author.get_full_name() if exhibit.author else "Неизвестен")
    collection_name = (exhibit.collection.name if exhibit.collection else "Неизвестная коллекция")
    condition_label = (exhibit.condition.label if exhibit.condition else "Неизвестно")
    status = exhibit.get_status()
    age = exhibit.calculate_age()

    print(f"--- ID: {exhibit.id} ---")
    print(f"Название: {exhibit.name}")
    print(f"Автор: {author_name}")
    print(f"Коллекция: {collection_name}")
    print(f"Год создания: {exhibit.year} (возраст: {age} лет)")
    print(f"Стоимость: {exhibit.price} руб.")
    print(f"Состояние: {condition_label}")
    print(f"Доступен: {'да' if exhibit.is_available else 'нет'}")
    print(f"Статус: {status}\n")

def print_all_exhibits(exhibits: Dict[int, Exhibit]):
    if not exhibits:
        print("Список экспонатов пуст.")
        return
    for ex in exhibits.values():
        print_exhibit_info(ex)

from models.authors import create_author
from models.collections import create_collection
from models.conditions import DEFAULT_CONDITIONS
from models.exhibits import (
    create_exhibit, find_exhibits_by_author,
    filter_exhibits_by_condition, filter_exhibits_by_collection,
    sort_exhibits_by_price, get_total_price,
)
from storage import (
    insert_author, insert_collection, insert_exhibit,
    load_authors, load_collections, load_conditions, load_exhibits,
)


def _seed(db_conn):
    a = create_author("Иван", "Шишкин")
    a.id = insert_author(a)
    c = create_collection("Живопись")
    c.id = insert_collection(c)
    cond = DEFAULT_CONDITIONS[2]
    ex = create_exhibit("Утро в сосновом лесу", a, c, cond, 1889, 3_500_000.0, True)
    ex.id = insert_exhibit(ex)
    return a, c, cond, ex


def test_insert_and_load_exhibit(db_conn):
    a, c, cond, ex = _seed(db_conn)
    authors = load_authors()
    colls = load_collections()
    conds = load_conditions()
    exhibits = load_exhibits(authors, colls, conds)
    assert len(exhibits) == 1
    assert exhibits[ex.id].name == "Утро в сосновом лесу"


def test_price_check_constraint(db_conn):
    import psycopg
    import pytest
    a = create_author("A", "B"); a.id = insert_author(a)
    c = create_collection("C"); c.id = insert_collection(c)
    cond = DEFAULT_CONDITIONS[1]
    bad = create_exhibit("Bad", a, c, cond, 1900, -1.0, True)
    with pytest.raises(psycopg.errors.CheckViolation):
        insert_exhibit(bad)


def test_filter_by_condition(db_conn):
    a, c, cond, ex = _seed(db_conn)
    authors = load_authors()
    colls = load_collections()
    conds = load_conditions()
    exhibits = load_exhibits(authors, colls, conds)
    assert len(filter_exhibits_by_condition(exhibits, cond.id)) == 1
    assert len(filter_exhibits_by_condition(exhibits, 4)) == 0


def test_sort_and_total(db_conn):
    a = create_author("A", "A"); a.id = insert_author(a)
    c = create_collection("C"); c.id = insert_collection(c)
    cond = DEFAULT_CONDITIONS[1]
    for name, price in [("One", 100.0), ("Two", 50.0), ("Three", 300.0)]:
        ex = create_exhibit(name, a, c, cond, 2000, price, True)
        ex.id = insert_exhibit(ex)
    exhibits = load_exhibits(load_authors(), load_collections(), load_conditions())
    sorted_ex = sort_exhibits_by_price(exhibits)
    assert [e.price for e in sorted_ex] == [50.0, 100.0, 300.0]
    assert get_total_price(exhibits) == 450.0
